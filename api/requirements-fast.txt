# Опциональные зависимости быстрого пути инференса (ветка feat/fast-inference).
#
# Базовый образ (api/requirements.txt) их не ставит: сервис работает без них.
# Ставить только если включаешь ускорения:
#   pip install -r requirements.txt -r requirements-fast.txt
#
# Включается по частям, каждая часть деградирует мягко, если пакета нет:
#   EMBED_MODEL=onnx:/models/<файл>.onnx   -> ONNX Runtime (нужен onnxruntime)
#   PyTurboJPEG                            -> быстрый JPEG-декод (иначе PIL)
#   orjson                                 -> быстрая сериализация ответов
#
# PyTurboJPEG требует системную libturbojpeg; если её нет, fastimg.py
# сам падает на PIL — сборка не ломается.

onnx==1.17.0             # torch.onnx.export в scripts/export_onnx.py
onnxruntime==1.20.1      # графовые оптимизации, автовыбор провайдера (TensorRT/CUDA/CPU)
PyTurboJPEG==1.7.7       # декодирование JPEG в 2-4 раза быстрее PIL
orjson==3.10.12          # сериализация ответов
