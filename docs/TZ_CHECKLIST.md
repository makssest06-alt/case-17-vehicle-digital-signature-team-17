```bash
./scripts/check_reproducibility.sh            # решение поднимается из чистого клона
python scripts/validate_submission.py         # артефакты формально валидны
python research/eval_service_path.py          # метрики тем кодом, которым отвечает сервис

python research/ann_benchmark.py --report     # масштабирование до миллиона
```
